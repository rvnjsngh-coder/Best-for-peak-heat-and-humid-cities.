# Best-for-peak-heat-and-humid-cities.
Choose a lighter GSM so the extra fabric doesn't feel heavy.
Light colours reflect heat; dark colours absorb it.

def pick_tshirt(temp_c, humid=False):
    if temp_c >= 38:
        return "160 GSM combed cotton, regular fit, light colour"
    if temp_c >= 32 or humid:
        return "180 GSM bio-washed cotton, relaxed fit"
    return "220 GSM cotton, oversized fit"

print(pick_tshirt(40))
